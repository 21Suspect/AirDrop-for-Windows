cmake_minimum_required(VERSION 3.20)
project(opendrop_intel_firmware C)
set(CMAKE_C_STANDARD 11)
add_library(iwl_fw_test SHARED iwl_firmware.c iwl_hw.c iwl_gen3.c iwl_pnvm.c iwl_command.c iwl_metadata.c iwl_response.c iwl_packet.c iwl_radio.c)
target_compile_definitions(iwl_fw_test PRIVATE IWL_FW_TEST_DLL)
add_executable(firmware-inspect firmware_inspect.c iwl_firmware.c)
if(MSVC)
  target_compile_options(iwl_fw_test PRIVATE /W4 /WX /GS /guard:cf)
  target_compile_options(firmware-inspect PRIVATE /W4 /WX /GS /guard:cf)
  target_compile_definitions(firmware-inspect PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(iwl_fw_test PRIVATE -Wall -Wextra -Werror)
  target_compile_options(firmware-inspect PRIVATE -Wall -Wextra -Werror)
endif()
